2. DALO (Insert at the end of the AssociateManager class, around line 183, before acc_manager = AssociateManager()):
```python
    async def generate_global_monthly_report(self, month: int, year: int) -> dict:
        """
        Retrieves sales/bookings across the entire system for a specific month/year.
        Only accessible by Admin/Super Admin users.
        """
        cursor = db.sales.find({"year": year, "month": month})
        sales = await cursor.to_list(length=None)
        
        # Convert ObjectId to string for JSON serialization
        formatted_sales = []
        for s in sales:
            s_copy = s.copy()
            if "_id" in s_copy:
                s_copy["_id"] = str(s_copy["_id"])
            if "user_id" in s_copy:
                s_copy["user_id"] = str(s_copy["user_id"])
            formatted_sales.append(s_copy)
            
        return {
            "sales": formatted_sales,
            "month": month,
            "year": year,
            "global": True
        }
```
